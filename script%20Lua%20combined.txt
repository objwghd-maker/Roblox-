    SpeedContainer.BorderSizePixel = 0

    ToggleBtn.Text = "SPEED 35 : OFF"
    ToggleBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    ToggleBtn.TextSize = 14
    ToggleBtn.Font = Enum.Font.GothamBold
    ToggleBtn.Active = true
    ToggleBtn.Parent = SpeedContainer

    local BtnCorner = Instance.new("UICorner")
    BtnCorner.CornerRadius = UDim.new(0, 8)
    BtnCorner.Parent = ToggleBtn

    local BtnStroke = Instance.new("UIStroke")
    BtnStroke.Color = Color3.fromRGB(0, 200, 255)
    BtnStroke.Thickness = 2
    BtnStroke.Parent = ToggleBtn

    local speedEnabled = false
    local defaultSpeed = 16
    local targetSpeed = 35
    local speedConnection = nil

    ToggleBtn.Activated:Connect(function()
        speedEnabled = not speedEnabled

        if speedEnabled then
            ToggleBtn.Text = "SPEED 35 : ON"
            TweenService:Create(ToggleBtn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(10, 50, 35)}):Play()
            TweenService:Create(BtnStroke, TweenInfo.new(0.2), {Color = Color3.fromRGB(0, 255, 150)}):Play()

            speedConnection = RunService.Heartbeat:Connect(function()
                local char = LocalPlayer.Character
                if char and char:FindFirstChildOfClass("Humanoid") then
                    char:FindFirstChildOfClass("Humanoid").WalkSpeed = targetSpeed
                end
            end)
        else
            ToggleBtn.Text = "SPEED 35 : OFF"
            TweenService:Create(ToggleBtn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(15, 20, 30)}):Play()
            TweenService:Create(BtnStroke, TweenInfo.new(0.2), {Color = Color3.fromRGB(0, 200, 255)}):Play()

                speedConnection:Disconnect()
                speedConnection = nil

            if char and char:FindFirstChildOfClass("Humanoid") then
                char:FindFirstChildOfClass("Humanoid").WalkSpeed = defaultSpeed
            end
        end

    local dragStartPos, startPos

    SpeedHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStartPos = input.Position
            startPos = SpeedContainer.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStartPos
            SpeedContainer.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end)

            playerGui.DeltaDashGui:Destroy()

        screenGui.Name = "DeltaDashGui"
        screenGui.ResetOnSpawn = false
        screenGui.Parent = playerGui

        local mainFrame = Instance.new("Frame")
        mainFrame.Name = "MainFrame"
        mainFrame.Size = UDim2.new(0, 100, 0, 116)
        mainFrame.Position = UDim2.new(0.8, 0, 0.5, -58)
        mainFrame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
        mainFrame.BorderSizePixel = 0
        mainFrame.Active = true
        mainFrame.Draggable = true
        mainFrame.Parent = screenGui

        local uiCornerMain = Instance.new("UICorner")
        uiCornerMain.CornerRadius = UDim.new(0, 8)
        uiCornerMain.Parent = mainFrame

        local uiStrokeMain = Instance.new("UIStroke")

        uiStrokeMain.Color = Color3.fromRGB(255, 255, 255)
        uiStrokeMain.Parent = mainFrame

        speedInput.Parent = mainFrame

        local uiCornerTop = Instance.new("UICorner")
        uiCornerTop.CornerRadius = UDim.new(0, 8)
        uiCornerTop.Parent = speedInput

        local dashButton = Instance.new("TextButton")
        dashButton.Name = "DashButton"
        dashButton.Size = UDim2.new(1, 0, 0, 40)
        dashButton.Position = UDim2.new(0, 0, 0, 28)
        dashButton.BackgroundColor3 = Color3.fromRGB(220, 50, 50)
        dashButton.TextColor3 = Color3.fromRGB(255, 255, 255)
        dashButton.TextSize = 15
        dashButton.Font = Enum.Font.SourceSansBold
        dashButton.Text = "🥱👍🤝"
        dashButton.BorderSizePixel = 0
        dashButton.Parent = mainFrame

        local lockButton = Instance.new("TextButton")
        lockButton.Name = "LockButton"
        lockButton.Size = UDim2.new(1, 0, 0, 28)

        uiCornerBot.Parent = lockButton

        local function toggleMenu()
            isExpanded = not isExpanded
            local targetSize = isExpanded and 116 or 20
            local tweenInfo = TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
            local sizeTween = TweenService:Create(mainFrame, tweenInfo, {Size = UDim2.new(0, 100, 0, targetSize)})
            sizeTween:Play()

            speedInput.Visible = isExpanded
            dashButton.Visible = isExpanded
            lockButton.Visible = isExpanded
        end

        local dragging = false
        local dragStartPos = nil
        local mainFrameStartPos = nil

        handle.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then

                dragStartPos = input.Position

                    mainFrameStartPos.X.Offset + delta.X,
                    mainFrameStartPos.Y.Scale,
                    mainFrameStartPos.Y.Offset + delta.Y

        handle.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
                local distance = (input.Position - dragStartPos).Magnitude
                if distance < 10 then
                    toggleMenu()
                end
                dragging = false
            end
        end)

        local isDashing = false
        local isLocked = false
        local dashSpeed = 70

        local savedWalkSpeed = 16
        local savedJumpPower = 50
        local savedAutoRotate = true

        local animateScript
        local animationStates = {}

        local dashConnection = nil
        local lockConnection = nil

        local currentAttachment = nil
        local currentLinearVelocity = nil
        local currentAntiGravity = nil

        local lockedCFrame = nil

        local function captureDefaults(humanoid)
            if humanoid then
                if humanoid.WalkSpeed > 0 then savedWalkSpeed = humanoid.WalkSpeed end
                if humanoid.JumpPower > 0 then savedJumpPower = humanoid.JumpPower end
                savedAutoRotate = humanoid.AutoRotate
            end
        end

        local function restoreState()
            local character = player.Character
            if character then
                local humanoid = character:FindFirstChildOfClass("Humanoid")
                if humanoid then
                    humanoid.WalkSpeed = savedWalkSpeed
                    humanoid.JumpPower = savedJumpPower
                    humanoid.AutoRotate = savedAutoRotate

                    humanoid:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
                    humanoid:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
                    humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
                end
            end
        end

        speedInput.FocusLost:Connect(function()
            local val = tonumber(speedInput.Text)
            if val then
                dashSpeed = val
            else
                speedInput.Text = ""
            end
        end)

        local function stopAnimations(character)
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                for _, track in ipairs(humanoid:GetPlayingAnimationTracks()) do
                    animationStates[track] = track.IsPlaying
                    track:Stop(0)
                end
            end
            animateScript = character:FindFirstChild("Animate")
            if animateScript then animateScript.Disabled = true end
        end

        local function restoreAnimations(character)
            if animateScript and animateScript.Parent == character then
                animateScript.Disabled = false
            end
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            if humanoid then
                for track, wasPlaying in pairs(animationStates) do
                    if track and track.Parent and wasPlaying then
                        pcall(function() track:Play(0) end)
                    end
                end
            end
            animationStates = {}
            animateScript = nil
        end

        local function stopDash()
            if dashConnection then
                dashConnection:Disconnect()
                dashConnection = nil
            end

            if currentLinearVelocity then currentLinearVelocity:Destroy() currentLinearVelocity = nil end
            if currentAntiGravity then currentAntiGravity:Destroy() currentAntiGravity = nil end
            if currentAttachment then currentAttachment:Destroy() currentAttachment = nil end

            isDashing = false
            dashButton.Text = "🥱👍🤝"
            dashButton.BackgroundColor3 = Color3.fromRGB(220, 50, 50)

            local character = player.Character
            if character then restoreAnimations(character) end

            if not isLocked then
                restoreState()
            end
        end

        local function startDash()
            if isDashing then
                stopDash()
                return
            end

            local character = player.Character
            if not character then return end

            local humanoid = character:FindFirstChildOfClass("Humanoid")
            local hrp = character:FindFirstChild("HumanoidRootPart")
            if not humanoid or not hrp then return end

            if not isLocked then
                captureDefaults(humanoid)
            end

            isDashing = true
            dashButton.Text = "🥱👍🤝"
            dashButton.BackgroundColor3 = Color3.fromRGB(150, 20, 20)

            humanoid.WalkSpeed = 0
            humanoid.JumpPower = 0

            humanoid:SetStateEnabled(Enum.HumanoidStateType.Jumping, false)
            humanoid:SetStateEnabled(Enum.HumanoidStateType.Freefall, false)
            humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)

            stopAnimations(character)

            currentAttachment = Instance.new("Attachment")
            currentAttachment.Parent = hrp

            currentLinearVelocity = Instance.new("LinearVelocity")
            currentLinearVelocity.Attachment0 = currentAttachment
            currentLinearVelocity.RelativeTo = Enum.ActuatorRelativeTo.World
            currentLinearVelocity.MaxForce = math.huge
            currentLinearVelocity.Parent = hrp

            currentAntiGravity = Instance.new("VectorForce")
            currentAntiGravity.Attachment0 = currentAttachment
            currentAntiGravity.RelativeTo = Enum.ActuatorRelativeTo.World
            currentAntiGravity.ApplyAtCenterOfMass = true
            currentAntiGravity.Force = Vector3.new(0, hrp.AssemblyMass * workspace.Gravity, 0)
            currentAntiGravity.Parent = hrp

            local fixedY = hrp.Position.Y

            dashConnection = RunService.PreRender:Connect(function()
                if not isDashing then return end

                local char = player.Character
                if not char or not char.Parent then
                    stopDash()
                    return
                end

                local hum = char:FindFirstChildOfClass("Humanoid")
                local root = char:FindFirstChild("HumanoidRootPart")
                if not hum or hum.Health <= 0 or not root then
                    stopDash()
                    return
                end

                hum.WalkSpeed = 0
                hum.JumpPower = 0

                for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
                    track:Stop(0)
                end

                local moveDir = hum.MoveDirection
                local targetDirection = (moveDir.Magnitude > 0) and moveDir or root.CFrame.LookVector

                local heightDiff = fixedY - root.Position.Y

                if currentLinearVelocity and currentLinearVelocity.Parent then
                    currentLinearVelocity.VectorVelocity = Vector3.new(
                        targetDirection.X * dashSpeed,
                        heightDiff * 15,
                        targetDirection.Z * dashSpeed
                    )
                end

                if currentAntiGravity and currentAntiGravity.Parent then
                    currentAntiGravity.Force = Vector3.new(0, root.AssemblyMass * workspace.Gravity, 0)
                end
            end)
        end

        dashButton.MouseButton1Click:Connect(startDash)

        local function stopLock()
            if lockConnection then
                lockConnection:Disconnect()
                lockConnection = nil
            end

            isLocked = false
            lockButton.BackgroundColor3 = Color3.fromRGB(0, 0, 0)

            if not isDashing then
                restoreState()
            end
        end

        local function toggleLock()
            if isLocked then
                stopLock()
                return
            end

            local character = player.Character
            if not character then return end
            local hrp = character:FindFirstChild("HumanoidRootPart")
            local humanoid = character:FindFirstChildOfClass("Humanoid")
            if not hrp or not humanoid then return end

            if not isDashing then
                captureDefaults(humanoid)
            end

            isLocked = true
            lockButton.BackgroundColor3 = Color3.fromRGB(0, 180, 80)
            lockedCFrame = hrp.CFrame

            lockConnection = RunService.PreRender:Connect(function()
                if not isLocked then return end

                local char = player.Character
                if not char then return end
                local root = char:FindFirstChild("HumanoidRootPart")
                local hum = char:FindFirstChildOfClass("Humanoid")

                if root and hum then
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero

                    hum.WalkSpeed = 0
                    hum.JumpPower = 0

                    hum:ChangeState(Enum.HumanoidStateType.Running)

                    local currentLook = root.CFrame.LookVector
                    lockedCFrame = CFrame.new(lockedCFrame.Position, lockedCFrame.Position + currentLook)
                    root.CFrame = lockedCFrame
                end
            end)
        end

        lockButton.MouseButton1Click:Connect(toggleLock)
    end)
end)

        local dragStart, startPos

        local function enableAntiFling()
            if connection then connection:Disconnect() end
            connection = RunService.Stepped:Connect(function()
                local myChar = lp.Character
                if not myChar then return end

                local hrp = myChar:FindFirstChild("HumanoidRootPart")
                if hrp then
                    if hrp.AssemblyLinearVelocity.Magnitude > 30 or hrp.AssemblyAngularVelocity.Magnitude > 20 then
                        hrp.AssemblyLinearVelocity = Vector3.zero
                        hrp.AssemblyAngularVelocity = Vector3.zero
                    end
                end

                for _, player in ipairs(Players:GetPlayers()) do
                    if player ~= lp and player.Character then
                        for _, part in ipairs(player.Character:GetChildren()) do
                            if part:IsA("BasePart") then
                                part.CanCollide = false
                            end
                        end
                    end
                end
            end)
        end

        local function disableAntiFling()
            if connection then
                connection:Disconnect()
                connection = nil
            end
        end

        Button.MouseButton1Click:Connect(function()
            antiFlingEnabled = not antiFlingEnabled
            if antiFlingEnabled then
                Button.Text = "Anti-Fling: ON"
                Button.BackgroundColor3 = Color3.fromRGB(50, 180, 50)
                enableAntiFling()
            else
                Button.Text = "Anti-Fling: OFF"
                Button.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
                disableAntiFling()
            end
        end)

        Button.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = Button.Position
            end
        end)

        Button.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end)

        UserInputService.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                local delta = input.Position - dragStart
                Button.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            end
        end)
    end)
end)

        local function addGradient(parent, colorA, colorB, rotation)
            local gradient = Instance.new("UIGradient")
            gradient.Color = ColorSequence.new(colorA, colorB)
            gradient.Rotation = rotation or 90
            gradient.Parent = parent
            return gradient
        end

        local function sharpenText(label)
            label.TextStrokeColor3 = Theme.TextOutline
            label.TextStrokeTransparency = 0.35
        end

        local oldScreenGui = LocalPlayer:WaitForChild("PlayerGui"):FindFirstChild("KillTrackerUI")

            oldScreenGui:Destroy()

            local data = activeTrackers[playerObj]

            if data.Att1 then data.Att1:Destroy() end

            if activeTrackers[playerObj] or not playerObj or playerObj.Parent ~= Players then
                return

            else
                Title.Text = "KILLERS LIST"

            for index, data in ipairs(killerList) do
                local playerObj = data.Player
                local killerName = data.Name
                local isActive = activeTrackers[playerObj] ~= nil

                local btn = Instance.new("TextButton")
                btn.Parent = ScrollingFrame
                btn.LayoutOrder = index
                btn.Size = UDim2.new(1, 0, 0, 26)
                btn.BackgroundColor3 = isActive and Theme.ButtonActive or Theme.ButtonIdle

                        tween(btn, {BackgroundColor3 = Theme.ButtonHover})

                btn.MouseLeave:Connect(function()
                    if not activeTrackers[playerObj] then
                        applyButtonState(btn, stroke, false)
                    end
                end)

                btn.Activated:Connect(function()
                    if activeTrackers[playerObj] then
                        removeSingleTracker(playerObj)
                        applyButtonState(btn, stroke, false)
                    elseif playerObj and playerObj.Parent == Players then
                        createSingleTracker(playerObj)
                        applyButtonState(btn, stroke, true)
                    end
                end)
            end
        end

            removeSingleTracker(player)

            for index = #killerList, 1, -1 do
                if killerList[index].Player == player then
                    table.remove(killerList, index)
                end
            end

            updateUI()
        end)

        local function getClosestPlayer(myChar)
            local closestPlayer
            local shortestDistance = 60
            local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")

            if not myHRP then
                return nil
            end

            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    local pHRP = p.Character:FindFirstChild("HumanoidRootPart")
                    local pHumanoid = p.Character:FindFirstChildOfClass("Humanoid")

                    if pHRP and pHumanoid and pHumanoid.Health > 0 then
                        local dist = (pHRP.Position - myHRP.Position).Magnitude

                        if dist < shortestDistance then
                            shortestDistance = dist
                            closestPlayer = p
                        end
                    end
                end
            end

            return closestPlayer
        end

        local function trackDeath(char)
            local humanoid = char:WaitForChild("Humanoid", 10)

            if not humanoid then
                return
            end

            humanoid.Died:Connect(function()
                local killerPlayer
                local creator = humanoid:FindFirstChild("creator")

                if creator and creator.Value and typeof(creator.Value) == "Instance" and creator.Value:IsA("Player") then
                    killerPlayer = creator.Value
                end

                if not killerPlayer then
                    killerPlayer = getClosestPlayer(char)
                end

                if not killerPlayer or killerPlayer == LocalPlayer or killerPlayer.Parent ~= Players then
                    return
                end

                for _, data in ipairs(killerList) do
                    if data.Player == killerPlayer then
                        return
                    end
                end

                table.insert(killerList, {
                    Player = killerPlayer,
                    Name = killerPlayer.Name
                })

                updateUI()
            end)
        end

        updateUI()

        if LocalPlayer.Character then
            task.spawn(trackDeath, LocalPlayer.Character)
        end

        LocalPlayer.CharacterAdded:Connect(trackDeath)
    end)
